{{/* Responsive with a WebP srcset. Every image on the site that needs srcset goes through here, so widths and quality live in one place. Params (dict): img image resource (required) alt alt text sizes how wide the image displays, e.g. "(min-width: 60rem) 29rem, 100vw" widths target widths (default 480 800 1280) loading "lazy" (default) or "eager" for images near the top of the page Never upscales: widths wider than the original are skipped, and an image narrower than the largest width also gets a copy at its own width. */}} {{ $img := .img }} {{ $widths := .widths | default (slice 480 800 1280) }} {{ $sizes := .sizes | default "100vw" }} {{ $loading := .loading | default "lazy" }} {{ $opts := "webp q80" }} {{ $srcset := slice }} {{ $largest := false }} {{ range $widths }} {{ if le . $img.Width }} {{ $r := $img.Resize (printf "%dx %s" . $opts) }} {{ $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $r.Width) }} {{ $largest = $r }} {{ end }} {{ end }} {{ $max := index $widths (sub (len $widths) 1) }} {{ if and (lt $img.Width $max) (not (in $widths $img.Width)) }} {{ $r := $img.Resize (printf "%dx %s" $img.Width $opts) }} {{ $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $r.Width) }} {{ $largest = $r }} {{ end }}
